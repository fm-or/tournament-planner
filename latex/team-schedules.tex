  \node[smallnode, #1] (G#2) {\strut #7};
    \node[smallnode, right=0mm of G#2] (G#2court) {\strut #5};
  \node[itemnode, right=0mm of G#2court] (G#2serve) {\strut #3};
  \node[itemnode,  right=0mm of G#2serve] (G#2receive) {\strut #4};

\DTLloaddb{scheduledata}{schedule.csv}% load the schedule data from csv

\DTLloaddb{groupdata}{groups.csv} % load group data from csv

\begin{document}
\pagenumbering{gobble} % no page numbers

% loop over entries in the database 'groupdata', extracting a list of teams
\makeatletter
\let\teamNames\empty
\DTLforeach{groupdata}{\group=Group,\team=Team}{%
  \IfSubStr{\teamNames}{\team}{}{% if not present
    \ifx\teamNames\empty
      \edef\teamNames{\team}%
    \else
      \edef\teamNames{\teamNames,\team}%
    \fi
  }%
}
\makeatother

\newcounter{zeitslotzaehler}

% loop through each team and print its schedule
\foreach \team in \teamNames {%
\setcounter{zeitslotzaehler}{0}

% print header for the schedule of the current team
\begin{center}
		\huge{\textbf{\tournamentname}}\\
		\vspace{1.25em}
		\LARGE{\textbf{Schedule of \team}}\\
\end{center}

\begin{figure}[H]
 \centering
   \begin{tikzpicture}
   \def\smallgap{5mm}
   \def\mediumgap{10mm}
   \def\largegap{15mm}
        	
   \coordinate (origin) at (0,0);

% print header
\createGame{below=0mm of origin}{Nr}{Serving}{Receiving}{Court}{Referee}{Time}{Points}% this is the header


\let\prevTime\empty
\let\timeOfLastMatch\empty

% loop through all matches
\DTLforeach{scheduledata}{\match=Match Nr,\court=Court,\serving=Team 1,\receiving=Team 2,\referee=Referee,\time=Time}
{% 

% check whether there was no match involving the current team in the previous slot
\ifthenelse{\not\equal{\prevTime}{\time} \AND \not\equal{\prevTime}{\timeOfLastMatch}}
{

% if this is the case, print an empty row containing only the time of the previous slot.
\stepcounter{zeitslotzaehler}
\ifnum\value{zeitslotzaehler}>1
   \expandafter\createGame
    \expandafter{below=0mm of G\number\numexpr\value{zeitslotzaehler}-1\relax}{\arabic{zeitslotzaehler}}{}{}{}{}{\prevTime}{}%
\else
    \expandafter\createGame
    \expandafter{below=0mm of GNr}{\arabic{zeitslotzaehler}}{}{}{}{}{\prevTime}{}%
\fi
}
{
%
}

% check if current match involves the current team
  \ifthenelse{\equal{\serving}{\team} \OR \equal{\receiving}{\team} \OR \equal{\referee}{\team}}{%
  
\stepcounter{zeitslotzaehler} % count up first
\global\let\timeOfLastMatch\time% update time of last match

% define entries to highlight the current team in bold
\def\printServing{\serving}
\def\printReceiving{\receiving}
\def\printReferee{\referee}
\ifthenelse{\equal{\serving}{\team}}{%
	\def\printServing{\textbf{\serving}}
}{%
   % 
}%
\ifthenelse{\equal{\receiving}{\team}}{%
	\def\printReceiving{\textbf{\receiving}}
}{%
   % 
}%
\ifthenelse{\equal{\referee}{\team}}{%
	\def\printReferee{\textbf{\referee}}
}{%
   % 
}%

% print match
 \ifnum\value{zeitslotzaehler}>1
   \expandafter\createGame
    \expandafter{below=0mm of G\number\numexpr\value{zeitslotzaehler}-1\relax}{\arabic{zeitslotzaehler}}{\printServing}{\printReceiving}{Court \court}{\printReferee}{\time}{:}%
\else
    \expandafter\createGame
    \expandafter{below=0mm of GNr}{\arabic{zeitslotzaehler}}{\printServing}{\printReceiving}{Court \court}{\printReferee}{\time}{:}%
\fi

  }{%
    % 
  }%
\global\let\prevTime\time% the time of the previous row is updated
}%

% if in the last slot there was no match where the current team was involved, an additional row must be printed
\ifthenelse{\not\equal{\prevTime}{\timeOfLastMatch}}
{
\stepcounter{zeitslotzaehler}
\ifnum\value{zeitslotzaehler}>1
   \expandafter\createGame
    \expandafter{below=0mm of G\number\numexpr\value{zeitslotzaehler}-1\relax}{\arabic{zeitslotzaehler}}{}{}{}{}{\prevTime}{}%
\else
    \expandafter\createGame
    \expandafter{below=0mm of GNr}{\arabic{zeitslotzaehler}}{}{}{}{}{\prevTime}{}%
\fi
}
{
%
}

\end{tikzpicture}
\end{figure}
